/// Live position of one unit that holds the job (incidents/{id}/tracks/{responderId}).
class TrackPoint {
  const TrackPoint({required this.latitude, required this.longitude, this.heading, this.speed, this.at});
  final double latitude;
  final double longitude;
  final double? heading;
  final double? speed;
  final DateTime? at;

  static TrackPoint? tryParse(Map<String, dynamic>? m) {
    final la = _d(m?['lat']);
    final lo = _d(m?['lng']);
    if (la == null || lo == null) return null;
    final at = m?['at'];
    return TrackPoint(latitude: la, longitude: lo, heading: _d(m?['heading']), speed: _d(m?['speed']), at: at is Timestamp ? at.toDate() : (at is DateTime ? at : null));
  }
}

/// Decodes a Google encoded polyline (precision 5). Returns [] for empty or malformed input.
List<({double lat, double lng})> decodePolyline(String encoded) {
  final out = <({double lat, double lng})>[];
  var index = 0, lat = 0, lng = 0;
  try {
    while (index < encoded.length) {
      for (var axis = 0; axis < 2; axis++) {
        var shift = 0, result = 0, b = 0;
        do {
          b = encoded.codeUnitAt(index++) - 63;
          result |= (b & 0x1f) << shift;
          shift += 5;
        } while (b >= 0x20);
        final delta = (result & 1) != 0 ? ~(result >> 1) : (result >> 1);
        if (axis == 0) {
          lat += delta;
        } else {
          lng += delta;
        }
      }
      out.add((lat: lat / 1e5, lng: lng / 1e5));
    }
  } catch (_) {
    return const [];
  }
  return out;
}

/// Straight-line distance in km (used only for the arrival check on the phone).
double haversineKm(double la1, double lo1, double la2, double lo2) {
  const r = 6371.0;
  final dLa = (la2 - la1) * math.pi / 180, dLo = (lo2 - lo1) * math.pi / 180;
  final a = math.pow(math.sin(dLa / 2), 2) + math.cos(la1 * math.pi / 180) * math.cos(la2 * math.pi / 180) * math.pow(math.sin(dLo / 2), 2);
  return 2 * r * math.asin(math.sqrt(a));
}

  'clinic': PinStyle(Color(0xFFC2185B), Icons.medical_services, BitmapDescriptor.hueRose),

  'helpline': PinStyle(Color(0xFF6A1B9A), Icons.support_agent, BitmapDescriptor.hueViolet),

  'me': PinStyle(Color(0xFF1A73E8), Icons.circle, BitmapDescriptor.hueBlue),

String pinKeyFor(PlaceType t) => t.name;

class PinFactory {
  static final Map<String, BitmapDescriptor> _cache = {};

  /// Returns a custom pin or null (use [fallback] instead).
  static Future<BitmapDescriptor?> pin(String key, {bool big = false}) async {
    final id = big ? '$key!' : key;
    final hit = _cache[id];
    if (hit != null) return hit;
    final style = kPinStyles[key];
    if (style == null) return null;
    try {
      final logical = big ? 54.0 : 40.0;
      const scale = 2.0;
      final size = logical * scale;
      final rec = ui.PictureRecorder();
      final c = Canvas(rec);
      final center = Offset(size / 2, size / 2);
      c.drawCircle(center + const Offset(0, 2), size / 2 - 3, Paint()..color = const Color(0x33000000));
      c.drawCircle(center, size / 2 - 3, Paint()..color = Colors.white);
      c.drawCircle(center, size / 2 - 8, Paint()..color = style.color);
      if (key != 'me') {
        final tp = TextPainter(
          text: TextSpan(text: String.fromCharCode(style.icon.codePoint), style: TextStyle(fontSize: size * 0.48, fontFamily: style.icon.fontFamily, package: style.icon.fontPackage, color: Colors.white)),
          textDirection: TextDirection.ltr,
        )..layout();
        tp.paint(c, center - Offset(tp.width / 2, tp.height / 2));
      }
      final img = await rec.endRecording().toImage(size.toInt(), size.toInt());
      final data = await img.toByteData(format: ui.ImageByteFormat.png);
      if (data == null) return null;
      final bytes = Uint8List.view(data.buffer);
      final d = BitmapDescriptor.bytes(bytes, width: logical, height: logical);
      _cache[id] = d;
      return d;
    } catch (_) {
      return null;
    }
  }

  static BitmapDescriptor fallback(String key) => BitmapDescriptor.defaultMarkerWithHue((kPinStyles[key] ?? kPinStyles['hospital']!).hue);
}

/// The four staff states of a unit, in words.
String unitGroup(String availability, DateTime? lastSeenAt, {DateTime? now}) {
  final seen = lastSeenAt;
  final stale = seen != null && (now ?? DateTime.now()).difference(seen).inSeconds > 120;
  if (stale) return 'Stale GPS';
  return switch (availability) {
    'Available' => 'Available',
    'Busy' => 'On a job',
    _ => 'Offline',
  };
}

        TrackPublisher.instance.sync(
          feed.offers,
        ); // live location only while a job is held
        for (final o in ResponderOfferQueue.openOffers(feed.offers)) {
          if (o.isOpen && o.secondsLeft() > 0 && _shown.add(o.id)) {
            if (_incomingOpen) {
              _waitingOfferId ??= o.id;
            } else {
              unawaited(_presentIncoming(o));
            }
            break;
          }
        }
      },
      onError: (Object _) {
        if (mounted) {
          setState(() {
            _offersError = true;
            _offersLoading = false;
          });
        }
      },
    );
  }

        offer = item;
        break;
      }
    }

  /// Call with the responder's full offers list every time it changes.

    final want = <String>{for (final o in offers) if (o.status == 'accepted' || o.status == 'enroute' || o.status == 'arrived') o.incidentId};
    if (setEquals(want, _incidents)) return;
    _incidents = want;

      stop();
    } else if (_sub == null) {
      _start();
    } else {
      _write(_last); // a second job started: publish immediately for it too
    }
  }

  Future<void> _start() async {

      if (p == LocationPermission.denied) p = await Geolocator.requestPermission();
      if (p == LocationPermission.denied || p == LocationPermission.deniedForever || !await Geolocator.isLocationServiceEnabled()) {
        problem.value = 'Turn on location so the person you are helping can see you coming.';
        return;
      }
      final LocationSettings settings = defaultTargetPlatform == TargetPlatform.android
          ? AndroidSettings(
              accuracy: LocationAccuracy.high,
              distanceFilter: 20,
              intervalDuration: const Duration(seconds: 5),
              foregroundNotificationConfig: const ForegroundNotificationConfig(
                notificationTitle: 'Emergency job in progress',
                notificationText: 'Sharing your live location with the person you are helping.',
                enableWakeLock: true,
              ),
            )
          : const LocationSettings(accuracy: LocationAccuracy.high, distanceFilter: 20);
      _sub = Geolocator.getPositionStream(locationSettings: settings).listen((pos) {
        _last = pos;
        _write(pos);
      }, onError: (Object _) => problem.value = 'Lost your location signal.');
      _keepAlive = Timer.periodic(const Duration(seconds: 15), (_) => _write(_last)); // standing still still counts as "live"
      problem.value = null;
    } catch (_) {
      problem.value = 'Could not start live location.';
    }
  }

  Future<void> _write(Position? pos) async {
    final uid = FirebaseAuth.instance.currentUser?.uid;
    if (pos == null || uid == null) return;
    for (final iid in _incidents) {
      try {
        await FirebaseFirestore.instance.collection('incidents').doc(iid).collection('tracks').doc(uid).set({
          'lat': pos.latitude,
          'lng': pos.longitude,
          'heading': pos.heading,
          'speed': pos.speed,
          'accuracy': pos.accuracy,
          'at': FieldValue.serverTimestamp(),
        });
      } catch (_) {/* offline or just finished: the next fix tries again */}
    }
  }

  void stop() {
    _sub?.cancel();
    _sub = null;
    _keepAlive?.cancel();
    _keepAlive = null;
    _incidents = {};
    _last = null;
  }
}

  Widget _summaryCard(Offer o) {

          _summaryRow('Accepted to arrived', accepted, arrived),
          _summaryRow('Arrived to complete', arrived, done),
          if (feedback != null) ...[

  Widget _summaryRow(String label, DateTime? from, DateTime? to) {
    final raw = from == null || to == null
        ? null
        : to.difference(from).inSeconds;
    final seconds = raw == null || raw >= 0 ? raw : 0;
    return Padding(
      padding: const EdgeInsets.only(bottom: 6),
      child: Row(
        children: [
          Expanded(
            child: Text(
              label,
              style: const TextStyle(fontSize: 13.5, color: RC.soft),
            ),
          ),
          Text(
            seconds == null
                ? 'Not recorded'
                : seconds < 60
                ? '${seconds < 0 ? 0 : seconds} s'
                : '${seconds ~/ 60} min ${seconds % 60} s',
            style: TextStyle(
              fontSize: 13.5,
              fontWeight: FontWeight.w700,
              color: seconds == null ? RC.muted : RC.text,
            ),
          ),
        ],
      ),
    );
  }
