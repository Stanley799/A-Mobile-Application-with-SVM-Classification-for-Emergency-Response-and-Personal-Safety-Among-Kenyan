"""Google Routes API (server side). Only used for travel time and the route line. Every call is optional: if the key is missing or Google is slow,
the caller falls back to a straight-line estimate and the system keeps working. The key arrives as the environment variable MAPS_SERVER_KEY
(a Firebase secret). NEVER log the key."""
import json
import os
import urllib.request

MATRIX_URL = "https://routes.googleapis.com/distanceMatrix/v2:computeRouteMatrix"
ROUTE_URL = "https://routes.googleapis.com/directions/v2:computeRoutes"
TIMEOUT = 3.5


def _wp(lat, lng):
    return {"waypoint": {"location": {"latLng": {"latitude": float(lat), "longitude": float(lng)}}}}


def _secs(v):
    try:
        return float(str(v).rstrip("s"))
    except (TypeError, ValueError):
        return None


def _post(url, key, mask, body, http=None):
    if http is not None:
        return http(url, key, mask, body)
    req = urllib.request.Request(url, data=json.dumps(body).encode(), method="POST",
                                 headers={"Content-Type": "application/json", "X-Goog-Api-Key": key, "X-Goog-FieldMask": mask})
    with urllib.request.urlopen(req, timeout=TIMEOUT) as r:  # noqa: S310 - fixed https URL
        return json.loads(r.read().decode())


def matrix_seconds(key, origins, destinations, http=None):
    """origins/destinations: lists of (lat, lng). Returns {(i, j): seconds} for the pairs Google could route. {} on any failure."""
    if not key or not origins or not destinations or len(origins) * len(destinations) > 25:
        return {}
    try:
        body = {"origins": [_wp(*o) for o in origins], "destinations": [_wp(*d) for d in destinations], "travelMode": "DRIVE"}
        data = _post(MATRIX_URL, key, "originIndex,destinationIndex,duration,condition", body, http)
        out = {}
        for e in data if isinstance(data, list) else []:
            s = _secs(e.get("duration"))
            if s is not None and e.get("condition", "ROUTE_EXISTS") == "ROUTE_EXISTS":
                out[(int(e.get("originIndex", 0)), int(e.get("destinationIndex", 0)))] = s
        return out
    except Exception as e:  # noqa: BLE001 - never block an emergency because a map call failed
        print(f"[routes] matrix failed: {type(e).__name__}")
        return {}


def route(key, origin, dest, http=None):
    """Returns {'polyline': str, 'durationS': float, 'distanceM': int} or None."""
    if not key:
        return None
    try:
        body = {"origin": _wp(*origin)["waypoint"], "destination": _wp(*dest)["waypoint"], "travelMode": "DRIVE"}
        data = _post(ROUTE_URL, key, "routes.duration,routes.distanceMeters,routes.polyline.encodedPolyline", body, http)
        r = (data.get("routes") or [None])[0]
        if not r:
            return None
        return {"polyline": (r.get("polyline") or {}).get("encodedPolyline", ""), "durationS": _secs(r.get("duration")) or 0.0, "distanceM": int(r.get("distanceMeters", 0))}
    except Exception as e:  # noqa: BLE001
        print(f"[routes] route failed: {type(e).__name__}")
        return None


def key():
    """The secret may be the word "unset" when no server key has been created yet: then everything uses straight-line estimates."""
    k = os.environ.get("MAPS_SERVER_KEY", "").strip()
    return None if k.lower() in ("", "unset", "none", "placeholder") else k
"""Route line + ETA for one accepted unit -> the incident, cached so the Routes API is called rarely."""
from . import core, routes, store

CACHE_FRESH_S, CACHE_MAX_S, MOVED_M = 30, 180, 150


def route_for(db, incident_id, responder_id, uid, is_admin, now, key=None, http=None):
    isnap = db.collection("incidents").document(incident_id).get()
    if not isnap.exists:
        return {"ok": False, "reason": "not_found"}
    inc = store.incident_view(incident_id, isnap.to_dict())
    if not (is_admin or uid == inc["userId"] or uid == responder_id):
        return {"ok": False, "reason": "forbidden"}
    offer = db.collection("offers").document(f"{incident_id}_{responder_id}").get()
    if not offer.exists or offer.to_dict().get("status") not in ("accepted", "enroute", "arrived") or inc["lat"] is None:
        return {"ok": False, "reason": "not_assigned"}
    tr = db.collection("incidents").document(incident_id).collection("tracks").document(responder_id).get()
    pos = tr.to_dict() if tr.exists else None
    if pos is None:
        rs = db.collection("responders").document(responder_id).get()
        r = rs.to_dict() if rs.exists else {}
        pos = {"lat": r.get("currentLatitude"), "lng": r.get("currentLongitude")}
    if pos.get("lat") is None:
        return {"ok": False, "reason": "no_position"}
    cache_ref = db.collection("incidents").document(incident_id).collection("routes").document(responder_id)
    c = cache_ref.get()
    cd = c.to_dict() if c.exists else None
    if cd:
        age = now - (store.epoch(cd.get("at")) or 0)
        moved = core.dl.haversine_km(pos["lat"], pos["lng"], cd["fromLat"], cd["fromLng"]) * 1000
        if age < CACHE_FRESH_S or (age < CACHE_MAX_S and moved < MOVED_M):
            return {"ok": True, "polyline": cd["polyline"], "etaS": cd["etaS"], "distanceM": cd["distanceM"], "ageS": round(age), "cached": True}
    r = routes.route(key if key is not None else routes.key(), (pos["lat"], pos["lng"]), (inc["lat"], inc["lng"]), http)
    if r is None:
        km = core.dl.haversine_km(pos["lat"], pos["lng"], inc["lat"], inc["lng"])
        return {"ok": True, "polyline": "", "etaS": round((2 + km * 1.35 / 30 * 60) * 60), "distanceM": round(km * 1000), "estimate": True}
    cache_ref.set({"polyline": r["polyline"], "etaS": r["durationS"], "distanceM": r["distanceM"], "fromLat": pos["lat"], "fromLng": pos["lng"], "at": store.when(now)})
    return {"ok": True, "polyline": r["polyline"], "etaS": r["durationS"], "distanceM": r["distanceM"], "cached": False}
